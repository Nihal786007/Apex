// Motor controls enabled after Arm. Match supply to motor voltage/current
// ratings. Reversed from the previous sketch to match your reported mounting.
constexpr bool SERVO_REVERSED = false;  // false: LEFT increases pulse width
// Both were false when Forward physically moved the robot backward.
// Channel A must be the left motor, channel B the right motor.
constexpr bool LEFT_MOTOR_REVERSED = true;
constexpr bool RIGHT_MOTOR_REVERSED = true;
constexpr bool SENSOR_ON_SERVO = true;  // false ONLY for a fixed forward sensor
constexpr int SERVO_CENTER_US = 1500;   // adjust mechanically at Center first
constexpr int SERVO_TRAVEL_US = 600;    // default endpoints: 900 and 2100 us
constexpr int SERVO_SAFE_MIN_US = 800, SERVO_SAFE_MAX_US = 2200;
static_assert(SERVO_CENTER_US >= 1200 && SERVO_CENTER_US <= 1800,
              "Invalid servo center");
static_assert(SERVO_TRAVEL_US >= 100 && SERVO_TRAVEL_US <= 700,
              "Invalid servo travel");
constexpr int ENA = 25, IN1 = 26, IN2 = 27, IN3 = 14, IN4 = 12, ENB = 13;
constexpr int SERVO_PIN = 18, TRIG = 5, ECHO = 19, SDA_PIN = 21, SCL_PIN = 22;
constexpr uint32_t HOLD_TIMEOUT = 450, SENSOR_STALE = 350, SERVO_SETTLE = 600;
constexpr float STOP_CM = 25.0f, CLEAR_CM = 30.0f;
WebServer server(80);
U8G2_SH1106_128X64_NONAME_F_HW_I2C oled(U8G2_R0, U8X8_PIN_NONE);
SemaphoreHandle_t guard;
bool oledOK = false, pwmOK = false, armed = false, rangeValid = false,
     obstacle = true;
int pan = 0,
    speedValue = 0;  // motors start at zero PWM; pan -100=LEFT, +100=RIGHT
char requested = 'S', actual = 'S';
float distanceCm = 0;
uint32_t lastHold = 0, lastRange = 0, servoChanged = 0, lastSequence = 0;
String session, oledText = "APEX READY", reason = "Disarmed";

void lockState() { xSemaphoreTake(guard, portMAX_DELAY); }
void unlockState() { xSemaphoreGive(guard); }
void motorOff() {
  ledcWrite(ENA, 0);
  ledcWrite(ENB, 0);
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}
void trip(const String &why) {
  armed = false;
  requested = 'S';
  reason = why;
}
String forwardBlock(uint32_t now) {
  if (SENSOR_ON_SERVO && pan != 0) return "Center sensor to drive forward";
  if (SENSOR_ON_SERVO && uint32_t(now - servoChanged) < SERVO_SETTLE)
    return "Sensor settling";
  if (!rangeValid || uint32_t(now - lastRange) > SENSOR_STALE)
    return "No fresh ultrasonic echo";
  if (obstacle) return "Forward blocked: needs 30 cm clear distance";
  return "";
}
void driveSide(int a, int b, int value, bool reversed) {
  if (reversed) value = -value;
  digitalWrite(a, value > 0);
  digitalWrite(b, value < 0);
}
void motorTask(void *) {
  char previous = 'S';
  uint32_t changedAt = 0;
  for (;;) {
    lockState();
    uint32_t now = millis();
    if (armed && requested != 'S' && uint32_t(now - lastHold) > HOLD_TIMEOUT)
      trip("Hold timed out: re-arm");
    if (armed && requested == 'F') {
      String why = forwardBlock(now);
      if (why.length()) trip(why + ": re-arm");
    }
    char target = (armed && pwmOK && speedValue > 0) ? requested : 'S';
    if (target != previous) {
      motorOff();
      actual = 'S';
      previous = target;
      changedAt = now;
    }
    // Neutral interval before any change of motor direction.
    if (target == 'S') {
      motorOff();
      actual = 'S';
    } else if (uint32_t(now - changedAt) >= 40) {
      int left = 1, right = 1;
      if (target == 'B') left = right = -1;
      if (target == 'L') {
        left = -1;
        right = 1;
      }
      if (target == 'R') {
        left = 1;
        right = -1;
      }
      driveSide(IN1, IN2, left, LEFT_MOTOR_REVERSED);
      driveSide(IN3, IN4, right, RIGHT_MOTOR_REVERSED);
      ledcWrite(ENA, speedValue);
      ledcWrite(ENB, speedValue);
      actual = target;
    }
    unlockState();
    vTaskDelay(pdMS_TO_TICKS(10));
  }
}
void setPan(int value) {  // caller holds guard; servo position is not measured
  value = constrain(value, -100, 100);
  if (value != pan) {
    pan = value;
    servoChanged = millis();
    if (SENSOR_ON_SERVO) {
      rangeValid = false;
      obstacle = true;
    }
    trip("Servo moved: re-arm");
  }
  int pulse =
      SERVO_CENTER_US + (SERVO_REVERSED ? 1 : -1) * pan * SERVO_TRAVEL_US / 100;
  uint32_t us = constrain(pulse, SERVO_SAFE_MIN_US, SERVO_SAFE_MAX_US);
  ledcWrite(SERVO_PIN, (us * 65535UL + 10000) / 20000UL);
}
void sampleRange() {
  digitalWrite(TRIG, LOW);
  delayMicroseconds(3);
  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG, LOW);
  uint32_t echo =
      pulseIn(ECHO, HIGH, 25000UL);  // bounded; motor watchdog runs separately
  float cm = echo * 0.0343f / 2.0f;
  lockState();
  rangeValid = echo > 0 && cm >= 2 && cm <= 400;
  distanceCm = cm;
  lastRange = millis();
  if (!rangeValid || cm <= STOP_CM)
    obstacle = true;
  else if (cm >= CLEAR_CM)
    obstacle = false;
  unlockState();
}

String jsonString(const String &s) {
  String out = "\"";
  for (size_t i = 0; i < s.length(); i++) {
    char c = s[i];
    if (c == '"' || c == '\\') out += '\\';
    if ((uint8_t)c >= 32) out += c;
  }
  return out + '"';
}
void reply(int code, const String &s) {
  server.sendHeader("Cache-Control", "no-store");
  server.send(code, "text/plain", s);
}
bool authenticated() {
  return session.length() && server.arg("token") == session;
}
bool numberArg(const String &s, long &v) {
  if (!s.length() || s.length() > 10) return false;
  char *end;
  v = strtol(s.c_str(), &end, 10);
  return *end == 0;
}
void setupRoutes() {
  server.on("/", HTTP_GET, [] { server.send_P(200, "text/html", PAGE); });
  server.on("/claim", HTTP_POST, [] {
    lockState();
    trip("Connected: arm when ready");
    session = String(esp_random(), HEX) + String(esp_random(), HEX);
    lastSequence = 0;
    speedValue = 0;
    String result = session;
    unlockState();
    reply(200, result);
  });
  server.on("/cmd", HTTP_POST, [] {
    if (!authenticated()) {
      reply(403, "Reconnect controller");
      return;
    }
    long sequence = 0;
    if (!numberArg(server.arg("seq"), sequence) || sequence <= 0) {
      reply(400, "Bad sequence");
      return;
    }
    lockState();
    if (uint32_t(sequence) <= lastSequence) {
      unlockState();
      reply(409, "Stale command ignored");
      return;
    }
    lastSequence = sequence;
    String k = server.arg("kind"), v = server.arg("value"), error = "";
    long n = 0;
    if (k == "stop")
      trip("Stopped: re-arm");
    else if (k == "release") {
      requested = 'S';
      if (armed) reason = "Stopped";  // Preserve a latched fault explanation.
    } else if (k == "arm") {
      if (!pwmOK)
        error = "PWM initialization failed";
      else {
        armed = true;
        requested = 'S';
        reason = "Armed: hold to drive";
      }
    } else if (k == "drive") {
      if (!armed)
        error = reason.length() ? reason : "Disarmed: press Arm motors";
      else if (speedValue == 0) {
        error = "Set speed above zero";
        trip(error);
      } else if (v.length() != 1 || String("FBLR").indexOf(v[0]) < 0)
        error = "Invalid direction";
      else if (v == "F" && (error = forwardBlock(millis())).length())
        trip(error + ": re-arm");
      else {
        requested = v[0];
        lastHold = millis();
        reason = requested == 'F'   ? "Forward selected"
                 : requested == 'B' ? "Backward selected"
                 : requested == 'L' ? "Left turn selected"
                                    : "Right turn selected";
      }
    } else if (k == "speed" && numberArg(v, n) && n >= 0 && n <= 255) {
      speedValue = n;
    } else if (k == "pan" && numberArg(v, n) && n >= -100 && n <= 100) {
      setPan(n);
    } else if (k == "text") {
      oledText = "";
      for (size_t i = 0; i < v.length() && oledText.length() < 40; i++)
        if (v[i] >= 32 && v[i] <= 126) oledText += v[i];
    } else
      error = "Invalid command";
    if (error.length()) {
      if (k == "drive") trip(error);  // Every rejected drive stops/disarms.
      else reason = error;
    }
    unlockState();
    reply(error.length() ? 409 : 200, error.length() ? error : "OK");
  });
  server.on("/status", HTTP_GET, [] {
    if (!authenticated()) {
      reply(403, "Reconnect");
      return;
    }
    lockState();
    bool fresh = rangeValid && uint32_t(millis() - lastRange) <= SENSOR_STALE;
    String j = "{\"armed\":" + String(armed ? "true" : "false") +
               ",\"valid\":" + (fresh ? "true" : "false") +
               ",\"cm\":" + String(distanceCm, 1) +
               ",\"reason\":" + jsonString(reason) +
               ",\"motion\":" + jsonString(String(actual)) +
               ",\"forward\":" + jsonString(forwardBlock(millis())) +
               ",\"text\":" + jsonString(oledText) +
               ",\"oled\":" + (oledOK ? "true" : "false") + "}";
    unlockState();
    server.sendHeader("Cache-Control", "no-store");
    server.send(200, "application/json", j);
  });
  server.onNotFound([] { reply(404, "Not found"); });
}
void setup() {
  Serial.begin(115200);
  for (int p : {ENA, IN1, IN2, IN3, IN4, ENB}) {
    pinMode(p, OUTPUT);
    digitalWrite(p, LOW);
  }
  pinMode(TRIG, OUTPUT);
  digitalWrite(TRIG, LOW);
  pinMode(ECHO, INPUT);
  guard = xSemaphoreCreateMutex();
  if (!guard) {
    Serial.println("FATAL: no mutex");
    while (true) delay(1000);
  }
  // Classic ESP32 channels 0/1 share motor timer; channel 4 uses a separate
  // servo timer.
  bool a = ledcAttachChannel(ENA, 20000, 8, 0),
       b = ledcAttachChannel(ENB, 20000, 8, 1);
  bool c = ledcAttachChannel(SERVO_PIN, 50, 16, 4);
  pwmOK = a && b && c;
  motorOff();
  servoChanged = millis();
  setPan(0);
  if (!pwmOK) {
    Serial.println("FATAL: PWM setup failed");
    while (true) delay(1000);
  }
  if (xTaskCreate(motorTask, "motorSafety", 4096, nullptr, 3, nullptr) !=
      pdPASS) {
    motorOff();
    Serial.println("FATAL: safety task failed");
    while (true) delay(1000);
  }
  Wire.begin(SDA_PIN, SCL_PIN);
  Wire.setTimeOut(25);
  for (uint8_t address : {uint8_t(0x3C), uint8_t(0x3D)}) {
    Wire.beginTransmission(address);
    if (Wire.endTransmission() == 0) {
      oled.setI2CAddress(address << 1);
      oled.begin();
      Wire.setTimeOut(25);
      oledOK = true;
      break;
    }
  }
  Serial.println(oledOK ? "OLED OK" : "OLED NOT DETECTED");
  WiFi.mode(WIFI_AP);
  if (!WiFi.softAP("APEX-CONTROL", "Apex2026Robot")) {
    Serial.println("FATAL: AP failed");
    while (true) delay(1000);
  }
  setupRoutes();
  server.begin();
  Serial.println(
      "APEX TABLET READY\nWiFi: APEX-CONTROL\nPassword: Apex2026Robot");
  Serial.print("Open: http://");
  Serial.println(WiFi.softAPIP());
  Serial.println(
      "Motors start stopped. Match supply to motor rating before power-on.");
}
void loop() {
  static uint32_t sensorTick = 0, displayTick = 0;
  server.handleClient();
  uint32_t now = millis();
  if (uint32_t(now - sensorTick) >= 80) {
    sensorTick = now;
    sampleRange();
  }
  if (oledOK && uint32_t(now - displayTick) >= 250) {
    displayTick = now;
    lockState();
    String text = oledText, why = reason;
    String range = rangeValid ? String(distanceCm, 1) + " cm" : "NO ECHO";
    bool on = armed;
    unlockState();
    oled.clearBuffer();
    oled.setFont(u8g2_font_6x10_tf);
    oled.drawStr(0, 10, on ? "APEX - ARMED" : "APEX - STOPPED");
    oled.drawStr(0, 22, range.c_str());
    oled.drawStr(0, 34, text.substring(0, 20).c_str());
    oled.drawStr(0, 44, text.substring(20, 40).c_str());
    oled.drawStr(0, 60, why.substring(0, 21).c_str());
    oled.sendBuffer();
  }
  delay(1);
}
